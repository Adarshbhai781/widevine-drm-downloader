   309→class Reader:
   310→    def __init__(self, buffer, cursor=0):
   311→        self._buffer = buffer
   312→        self._cursor = cursor
   313→    def u1(self, move_cursor=True) -> int:
   314→        return self.unpack('B', move_cursor=move_cursor)[0]
   315→    def u4(self, move_cursor=True) -> int:
   316→        return self.unpack('<I', move_cursor=move_cursor)[0]
   317→    def u8(self, move_cursor=True) -> int:
   318→        return self.unpack('<Q', move_cursor=move_cursor)[0]
   319→    def i1(self, move_cursor=True) -> int:
   320→        return self.unpack('b', move_cursor=move_cursor)[0]
   321→    def i4(self, move_cursor=True) -> int:
   322→        return self.unpack('<i', move_cursor=move_cursor)[0]
   323→    def i8(self, move_cursor=True) -> int:
   324→        return self.unpack('<q', move_cursor=move_cursor)[0]
   325→    def s(self, n: int, move_cursor=True) -> bytes:
   326→        return self.unpack(f'{n}s', move_cursor=move_cursor)[0]
   327→    def unpack(self, f: str, offset=0, move_cursor=True):
   328→        x = struct.unpack_from(f, self._buffer, self._cursor + offset)
   329→        if move_cursor:
   330→            self._cursor += struct.calcsize(f)
   331→        return x
   332→    def string(self, move_cursor=True) -> str:
   333→        length = self.i4(move_cursor=move_cursor)
   334→        if length == 0:
   335→            return str()
   336→        else:
   337→            assert length > 0
   338→            offset = 0 if move_cursor else 4
   339→            return self.unpack(f'{length}s', offset=offset, move_cursor=move_cursor)[0].rstrip(b'\x00').decode()

   341→class PakInfo:
   342→    def __init__(self, buffer, keystream: List[int]):
   343→        def decrypt_index_encrypted(x: int) -> int:
   344→            MASK_8 = 255
   345→            return (x ^ keystream[3]) & MASK_8
   346→        def decrypt_magic(x: int) -> int:
   347→            return x ^ keystream[2]
   348→        def decrypt_index_hash(x: bytes) -> bytes:
   349→            key = struct.pack('<5I', *keystream[4:][:5])
   350→            assert len(x) == len(key)
   351→            return bytes((a ^ b for a, b in zip(x, key)))
   352→        def decrypt_index_size(x: int) -> int:
   353→            return x ^ (keystream[10] << 32 | keystream[11])
   354→        def decrypt_index_offset(x: int) -> int:
   355→            return x ^ (keystream[0] << 32 | keystream[1])
   356→        reader = Reader(buffer[-PakInfo._mem_size((-1)):])
   357→        self.index_encrypted = decrypt_index_encrypted(reader.u1()) == 1
   358→        self.magic = decrypt_magic(reader.u4())
   359→        self.version = reader.u4()
   360→        self.index_hash = decrypt_index_hash(reader.s(20)) if self.version >= 6 else bytes()
   361→        self.index_size = decrypt_index_size(reader.u8())
   362→        self.index_offset = decrypt_index_offset(reader.u8())
   363→        if self.version <= 3:
   364→            self.index_encrypted = False
   365→    @staticmethod
   366→    def _mem_size(_: int) -> int:
   367→        return 45

   401→@dataclass
   402→class TencentPakEntry:
   403→    def __init__(self, reader: Reader, version: int):
   404→        self.content_hash = reader.s(20)
   405→        if version <= 1:
   406→            _ = reader.u8()
   407→        self.offset = reader.u8()
   408→        self.uncompressed_size = reader.u8()
   409→        self.compression_method = reader.u4() & CM_MASK
   410→        self.size = reader.u8()
   411→        self.unk1 = reader.u1() if version >= 5 else 0
   412→        self.unk2 = reader.s(20) if version >= 5 else bytes()
   413→        if self.compression_method != 0 and version >= 3:
   414→            self.compressed_blocks = [PakCompressedBlock(reader) for _ in range(reader.u4())]
   415→        else:
   416→            self.compressed_blocks = []
   417→        self.compression_block_size = reader.u4() if version >= 4 else 0
   418→        self.encrypted = reader.u1() == 1 if version >= 4 else False
   419→        self.encryption_method = reader.u4() if version >= 12 else 0
   420→        self.index_new_sep = reader.u4() if version >= 12 else 0

   422→class PakCrypto:
   423→    class _LCG:
   424→        def __init__(self, seed: int):
   425→            self.state = seed
   426→        def next(self) -> int:
   427→            MASK_32 = 4294967295
   428→            MSB_1 = 2147483648
   429→            def wrap(x: int) -> int:
   430→                x &= MASK_32
   431→                if not x & MSB_1:
   432→                    return x
   433→                else:
   434→                    return (x + MSB_1 & MASK_32) - MSB_1
   435→            x1 = wrap(1103515245 * self.state)
   436→            self.state = wrap(x1 + 12345)
   437→            x2 = wrap(x1 + 77880) if self.state < 0 else self.state
   438→            return (x2 >> 16 & MASK_32) % 32767
   439→    @staticmethod
   440→    def zuc_keystream() -> List[int]:
   441→        zuc = gmalg.ZUC(ZUC_KEY, ZUC_IV)
   442→        return [struct.unpack('>I', zuc.generate())[0] for _ in range(16)]
   443→    @staticmethod
   444→    def _xorxor(buffer, x) -> bytes:
   445→        return bytes((buffer[i] ^ x[i % len(x)] for i in range(len(buffer))))
   446→    @staticmethod
   447→    def _hashhash(buffer, n: int) -> bytes:
   448→        result = bytes()
   449→        for i in range(math.ceil(n / SHA1.digest_size)):
   450→            result += SHA1.new(buffer).digest()
   451→        if len(result) >= n:
   452→            result = result[:n]
   453→            return result
   454→        else:
   455→            result += b'\x00' * (n - len(result))
   456→            return result
   457→    @staticmethod
   458→    def _meowmeow(buffer) -> bytes:
   459→        def unpad(x):
   460→            skip = 1 + next((i for i in range(len(x)) if x[i]!= 0))
   461→            return x[skip:]
   462→        if len(buffer) < 43:
   463→            return bytes()
   464→        else:
   465→            x1 = buffer[1:][:SHA1.digest_size]
   466→            x2 = buffer[SHA1.digest_size + 1:]
   467→            x1 = PakCrypto._xorxor(x1, PakCrypto._hashhash(x2, len(x1)))
   468→            x2 = PakCrypto._xorxor(x2, PakCrypto._hashhash(x1, len(x2)))
   469→            part1, m = (x2[:SHA1.digest_size], x2[SHA1.digest_size:])
   470→            if part1!= SHA1.new(b'\x00' * SHA1.digest_size).digest():
   471→                return bytes()
   472→            else:
   473→                return unpad(m)
   474→    @staticmethod
   475→    def rsa_extract(signature: bytes, modulus: bytes) -> bytes:
   476→        c = int.from_bytes(signature, 'little')
   477→        n = int.from_bytes(modulus, 'little')
   478→        e = 65537
   479→        m = pow(c, e, n).to_bytes(256, 'little').rstrip(b'\x00')
   480→        return PakCrypto._meowmeow(Misc.pad_to_n(m, 4))
   481→    @staticmethod
   482→    def _decrypt_simple1(ciphertext) -> bytes:
   483→        return bytes((x ^ SIMPLE1_DECRYPT_KEY for x in ciphertext))
   484→    @staticmethod
   485→    def _decrypt_simple2(ciphertext) -> bytes:
   486→        class RollingKey:
   487→            def __init__(self, initial_value: int):
   488→                self._value = initial_value
   489→            def update(self, x: int) -> int:
   490→                self._value ^= x
   491→                return self._value
   492→        assert len(ciphertext) % SIMPLE2_BLOCK_SIZE == 0
   493→        initial_key, = struct.unpack('<I', SIMPLE2_DECRYPT_KEY)
   494→        rolling_key = RollingKey(initial_key)
   495→        plaintext = (struct.pack('<I', rolling_key.update(x)) for x in struct.unpack(f'<{len(ciphertext) // 4}I', ciphertext))
   496→        return bytes(it.chain.from_iterable(plaintext))
   497→    @staticmethod
   498→    @lru_cache(maxsize=1)
   499→    def _derive_sm4_key(file_path: PurePath, encryption_method: int) -> bytes:
   500→        part1 = file_path.stem.lower()
   501→        if encryption_method == EM_SM4_2:
   502→            secret = SM4_SECRET_2
   503→        else:
   504→            if encryption_method == EM_SM4_4:
   505→                secret = SM4_SECRET_4
   506→            else:
   507→                index = (encryption_method - EM_SM4_NEW_BASE) % len(SM4_SECRET_NEW)
   508→                secret = f'{SM4_SECRET_NEW[index]}{encryption_method}'
   509→        return SHA1.new(str(part1 + secret).encode()).digest()[:SM4.key_length()]
   510→    @staticmethod
   511→    @lru_cache(maxsize=1)
   512→    def _sm4_context_for_key(key: bytes) -> SM4:
   513→        return SM4(key)
   514→    @staticmethod
   515→    def _decrypt_sm4(ciphertext, file_path: PurePath, encryption_method: int) -> bytes:
   516→        assert len(ciphertext) % SM4.block_length() == 0
   517→        key = PakCrypto._derive_sm4_key(file_path, encryption_method)
   518→        sm4 = PakCrypto._sm4_context_for_key(key)
   519→        return bytes(it.chain.from_iterable((sm4.decrypt(x) for x in it.batched(ciphertext, SM4.block_length()))))
   520→    @staticmethod
   521→    def decrypt_index(ciphertext, pak_info: TencentPakInfo) -> bytes:
   522→        if pak_info.version > 7:
   523→            key = PakCrypto.rsa_extract(pak_info.packed_key, RSA_MOD_1)
   524→            iv = PakCrypto.rsa_extract(pak_info.packed_iv, RSA_MOD_1)
   525→            assert len(key) == 32 and len(iv) == 32
   526→            aes = AES.new(key, MODE_CBC, iv[:16])
   527→            return unpad(aes.decrypt(ciphertext), AES.block_size)
   528→        else:
   529→            return bytes(PakCrypto._decrypt_simple1(ciphertext))
   530→    @staticmethod
   531→    def _is_simple1_method(encryption_method: int) -> bool:
   532→        return encryption_method == EM_SIMPLE1
   533→    @staticmethod
   534→    def _is_simple2_method(encryption_method: int) -> bool:
   535→        return encryption_method == EM_SIMPLE2 or encryption_method == 17
   536→    @staticmethod
   537→    def _is_sm4_method(encryption_method: int) -> bool:
   538→        return encryption_method == EM_SM4_2 or encryption_method == EM_SM4_4 or encryption_method & EM_SM4_NEW_MASK!= 0
   539→    @staticmethod
   540→    def align_encrypted_content_size(n: int, encryption_method: int) -> int:
   541→        if PakCrypto._is_simple2_method(encryption_method):
   542→            return Misc.align_up(n, SIMPLE2_BLOCK_SIZE)
   543→        else:
   544→            if PakCrypto._is_sm4_method(encryption_method):
   545→                return Misc.align_up(n, SM4.block_length())
   546→            else:
   547→                return n
   548→    @staticmethod
   549→    def decrypt_block(ciphertext, file: PurePath, encryption_method: int) -> bytes:
   550→        if PakCrypto._is_simple1_method(encryption_method):
   551→            return PakCrypto._decrypt_simple1(ciphertext)
   552→        else:
   553→            if PakCrypto._is_simple2_method(encryption_method):
   554→                return PakCrypto._decrypt_simple2(ciphertext)
   555→            else:
   556→                if PakCrypto._is_sm4_method(encryption_method):
   557→                    return PakCrypto._decrypt_sm4(ciphertext, file, encryption_method)
   558→                else:
   559→                    raise ValueError(f'Unknown encryption method: {encryption_method}')
   560→    @staticmethod
   561→    @lru_cache(maxsize=33)
   562→    def generate_block_indices(n: int, encryption_method: int) -> List[int]:
   563→        if not PakCrypto._is_sm4_method(encryption_method):
   564→            return list(range(n))
   565→        else:
   566→            permutation = []
   567→            lcg = PakCrypto._LCG(n)
   568→            while len(permutation)!= n:
   569→                x = lcg.next() % n
   570→                if x not in permutation:
   571→                    permutation.append(x)
   572→            inverse = [0] * len(permutation)
   573→            for i, x in enumerate(permutation):
   574→                inverse[x] = i
   575→            return inverse

   577→class PakCompression:
   578→    @staticmethod
   579→    @lru_cache(maxsize=33)
   580→    def _zstd_decompressor(dict: ZstdCompressionDict) -> ZstdDecompressor:
   581→        return ZstdDecompressor(dict)
   582→    @staticmethod
   583→    def zstd_dictionary(dict_data) -> ZstdCompressionDict:
   584→        return ZstdCompressionDict(dict_data, DICT_TYPE_AUTO)
   585→    @staticmethod
   586→    def decompress_block(block, dict: Optional[ZstdCompressionDict], compression_method: int) -> bytes:
   587→        if compression_method == CM_ZLIB:
   588→            try:
   589→                return zlib.decompress(block)
   590→            except zlib.error:
   591→                return block
   592→        else:
   593→            if compression_method == CM_ZSTD or compression_method == CM_ZSTD_DICT:
   594→                if compression_method!= CM_ZSTD_DICT:
   595→                    dict = None

   597→            else:
   598→                raise ValueError(f'Unknown compression method: {compression_method}')

   600→class TencentPakFile:
   601→    def __init__(self, file_path: PurePath, is_od=False):
   602→        self._file_path = file_path
   603→        with open(file_path, 'rb') as file:
   604→            self._file_content = memoryview(file.read())
   605→        self._is_od = is_od
   606→        self._mount_point = PurePath()
   607→        self._is_zstd_with_dict = 'zsdic' in str(self._file_path)
   608→        self._zstd_dict = None
   609→        self._files = []
   610→        self._index = {}
   611→        self._pak_info = TencentPakInfo(self._file_content, PakCrypto.zuc_keystream())
   612→        self._verify_stem_hash()
   613→        self._tencent_load_index()
   614→    
   615→    def _get_method_str(self, method_int, is_encryption):
   616→        if is_encryption:
   617→            if PakCrypto._is_simple1_method(method_int): return "SIMPLE1"
   618→            if PakCrypto._is_simple2_method(method_int): return "SIMPLE2"
   619→            if PakCrypto._is_sm4_method(method_int): return f"SM4 (Type {method_int})"
   620→            return "NONE" if method_int == 0 else "UNKNOWN"
   621→        else:
   622→            if method_int == CM_NONE: return "NONE"
   623→            if method_int == CM_ZLIB: return "ZLIB"
   624→            if method_int == CM_ZSTD: return "ZSTD"
   625→            if method_int == CM_ZSTD_DICT: return "ZSTD_DICT"
   626→            return "UNKNOWN"

   628→    def _verify_stem_hash(self) -> None:
   629→        if not self._is_od and self._pak_info.version >= 9:
   630→                assert self._pak_info.stem_hash == zlib.crc32(self._file_path.stem.encode('utf-32le'))
   631→    def _tencent_load_index(self) -> None:
   632→        index_data = self._file_content[self._pak_info.index_offset:][:self._pak_info.index_size]
   633→        if self._pak_info.index_encrypted:
   634→            index_data = PakCrypto.decrypt_index(index_data, self._pak_info)
   635→        else:
   636→            index_data = index_data
   637→        self._verify_index_hash(index_data)
   638→        self._load_index(index_data)
   639→    def _verify_index_hash(self, index_data) -> None:
   640→        expected_hash = self._pak_info.index_hash
   641→        if not self._is_od and self._pak_info.version >= 8:
   642→                assert expected_hash == PakCrypto.rsa_extract(self._pak_info.packed_index_hash, RSA_MOD_2)
   643→        assert expected_hash == SHA1.new(index_data).digest()
   644→    @staticmethod
   645→    def _construct_mount_point(mount_point: str) -> PurePath:
   646→        result = PurePath()
   647→        for part in PurePath(mount_point).parts:
   648→            if part!= '..':
   649→                result /= part
   650→        return result
   651→    def _peek_content(self, offset: int, size: int, encryption_method: int) -> memoryview:
   652→        size = PakCrypto.align_encrypted_content_size(size, encryption_method)
   653→        return self._file_content[offset:][:size]
   654→    def _peek_block_content(self, block: PakCompressedBlock, encryption_method: int) -> memoryview:
   655→        size = PakCrypto.align_encrypted_content_size(block.end - block.start, encryption_method)
   656→        return self._file_content[block.start:][:size]
   657→    def _construct_zstd_dict(self, dict_entry: TencentPakEntry) -> None:
   658→        assert not self._zstd_dict
   659→        assert not dict_entry.encrypted
   660→        assert dict_entry.compression_method == CM_NONE
   661→        reader = Reader(self._peek_content(dict_entry.offset, dict_entry.size, 0))
   662→        dict_size = reader.u8()
   663→        _ = reader.u4()
   664→        assert dict_size == reader.u4()
   665→        dict_data = reader.s(dict_size)
   666→        self._zstd_dict = PakCompression.zstd_dictionary(dict_data)
   667→    def _load_index(self, index_data) -> None:
   668→        if self._pak_info.version <= 10:
   669→            raise ValueError(f'Unsupported version: {self._pak_info.version}')
   670→        else:
   671→            reader = Reader(index_data)
   672→            self._mount_point = self._construct_mount_point(reader.string())
   673→            self._files = [TencentPakEntry(reader, self._pak_info.version) for _ in range(reader.u4())]
   674→            for _ in range(reader.u8()):
   675→                dir_path = PurePath(reader.string())
   676→                e = {reader.string(): self._files[~reader.i4()] for _ in range(reader.u8())}
   677→                if self._is_zstd_with_dict and dir_path.name == 'zstddic':
   678→                    assert len(e) == 1
   679→                    self._construct_zstd_dict(e[[*e.keys()][0]])
   680→                else:
   681→                    self._index.update({PurePath(dir_path): e})

   683→    def _write_to_disk(self, file_path: Path, entry: TencentPakEntry) -> None:
   684→        encryption_method = entry.encryption_method
   685→        compression_method = entry.compression_method
   686→
   687→        enc_str = self._get_method_str(encryption_method, True)
   688→        comp_str = self._get_method_str(compression_method, False)
   689→        console.print(f"[bold cyan]->[/] Unpack: [bold green]{file_path.name}[/] [[bold yellow]{comp_str}[/]/[bold magenta]{enc_str}[/]]")
   690→
   691→        with open(file_path, 'wb') as file:
   692→            if compression_method == CM_NONE:
   693→                data = self._peek_content(entry.offset, entry.size, encryption_method)
   694→                if entry.encrypted:
   695→                    data = PakCrypto.decrypt_block(data, file_path, encryption_method)
   696→                file.write(data)
   697→                return
   698→            else:
   699→                for x in PakCrypto.generate_block_indices(len(entry.compressed_blocks), encryption_method):
   700→                    data = self._peek_block_content(entry.compressed_blocks[x], encryption_method)
   701→                    if entry.encrypted:
   702→                        data = PakCrypto.decrypt_block(data, file_path, encryption_method)
   703→                    data = PakCompression.decompress_block(data, self._zstd_dict, compression_method)
   704→                    file.write(data)

   706→    def dump(self, out_path: Path) -> None:

   755→def _zstd_add_skippable_padding(data: bytes, pad_len: int) -> bytes:
   756→    if pad_len <= 0:
   757→        return data
   758→    else:
   759→        out = bytearray(data)
   760→        while pad_len > 0:
   761→            frame_len = min(max(pad_len - 8, 0), 1048576)
   762→            out += b'P*M\x18'
   763→            out += struct.pack('<I', frame_len)
   764→            out += b'\x00' * frame_len
   765→            pad_len -= 8 + frame_len
   766→        return bytes(out)

   768→def _encrypt_plaintext(plaintext: bytes, pak_relative_path: PurePath, encryption_method: int) -> bytes:
   769→    if PakCrypto._is_simple1_method(encryption_method):
   770→        return bytes((b ^ SIMPLE1_DECRYPT_KEY for b in plaintext))
   771→    else:
   772→        if PakCrypto._is_simple2_method(encryption_method):
   773→            pad = -len(plaintext) % SIMPLE2_BLOCK_SIZE
   774→            plaintext += b'\x00' * pad
   775→            key, = struct.unpack('<I', SIMPLE2_DECRYPT_KEY)
   776→            rolling = key
   777→            out = []
   778→            for x, in struct.iter_unpack('<I', plaintext):
   779→                c = rolling ^ x
   780→                out.append(c)
   781→                rolling ^= c
   782→            return struct.pack(f'<{len(out)}I', *out)
   783→        else:
   784→            if PakCrypto._is_sm4_method(encryption_method):
   785→                key = PakCrypto._derive_sm4_key(pak_relative_path, encryption_method)
   786→                sm4 = PakCrypto._sm4_context_for_key(key)
   787→                pad_len = -len(plaintext) % 16
   788→                if pad_len > 0:
   789→                    plaintext = plaintext + b'\x00' * pad_len
   790→                out = bytearray()
   791→                for i in range(0, len(plaintext), 16):
   792→                    block = plaintext[i:i + 16]
   793→                    if len(block) < 16:
   794→                        block = block.ljust(16, b'\x00')
   795→                    out.extend(sm4.encrypt(block))
   796→                return bytes(out)
   797→            else:
   798→                return plaintext

   802→def _repack_uncompressed(outfh, pak_file, entry, pak_relative_path: PurePath, new_data: bytes):
   803→    enc_method = entry.encryption_method
   804→    target_size = entry.size
   805→    enc_region = PakCrypto.align_encrypted_content_size(target_size, enc_method) if entry.encrypted else target_size
   806→    plaintext = new_data[:enc_region]
   807→    if entry.encrypted:
   808→        a = PakCrypto.align_encrypted_content_size(len(plaintext), enc_method)
   809→        plaintext += b'\x00' * (a - len(plaintext))
   810→        cipher = _encrypt_plaintext(plaintext, pak_relative_path, enc_method)
   811→        outfh.seek(entry.offset)
   812→        outfh.write(cipher)
   813→        with open(pak_file._file_path, 'rb') as src:
   814→            src.seek(entry.offset + len(cipher))
   815→            outfh.write(src.read(enc_region - len(cipher)))
   816→    else:
   817→        outfh.seek(entry.offset)
   818→        outfh.write(plaintext)
   819→        with open(pak_file._file_path, 'rb') as src:
   820→            src.seek(entry.offset + len(plaintext))
   821→            outfh.write(src.read(target_size - len(plaintext)))

   823→def _best_compress(chunk, cm, zstd_dict=None):
   824→    """Compress one chunk at the best achievable level."""
   825→    if cm == CM_ZLIB:
   826→        return zlib.compress(chunk, 9)
   827→    if cm in (CM_ZSTD, CM_ZSTD_DICT):
   828→        zd = zstd_dict if cm == CM_ZSTD_DICT else None
   829→        for lvl in [22, 19, 16, 13, 10, 7, 4, 1]:
   830→            try:
   831→                return ZstdCompressor(level=lvl, dict_data=zd, threads=1).compress(chunk)
   832→            except Exception:
   833→                continue
   834→    return chunk  # fallback: store raw

   836→def _pw_string(s):
   837→    """PAK string serialiser: i4(len_with_null) + bytes + null."""
   838→    if not s: return struct.pack('<i', 0)
   839→    b = s.encode('utf-8') + b'\x00'
   840→    return struct.pack('<i', len(b)) + b

   842→def _pw_entry(e, v):
   843→    """Serialise one TencentPakEntry back to bytes."""
   844→    w = bytearray(e.content_hash)
   845→    w += struct.pack('<Q', e.offset)
   846→    w += struct.pack('<Q', e.uncompressed_size)
   847→    w += struct.pack('<I', e.compression_method)
   848→    w += struct.pack('<Q', e.size)
   849→    if v >= 5:
   850→        w += bytes([e.unk1])
   851→        w += e.unk2  # 20 bytes
   852→    if e.compression_method != CM_NONE and v >= 3:
   853→        w += struct.pack('<I', len(e.compressed_blocks))
   854→        for b in e.compressed_blocks:
   855→            w += struct.pack('<QQ', b.start, b.end)
   856→    if v >= 4:
   857→        w += struct.pack('<I', e.compression_block_size)
   858→        w += bytes([1 if e.encrypted else 0])
   859→    if v >= 12:
   860→        w += struct.pack('<II', e.encryption_method, e.index_new_sep)
   861→    return bytes(w)

  1348→def smart_resolve_by_fingerprint(filename: str, repack_file: Path, candidates: list):
  1349→    repack_size = repack_file.stat().st_size
  1350→    size_matches = [(path, entry) for path, entry in candidates if entry.uncompressed_size == repack_size]
  1351→    if len(size_matches) == 1:
  1352→        return size_matches[0]
  1353→    if not size_matches:
  1354→        return None
  1355→    def fingerprint(e):
  1356→        return (e.uncompressed_size, e.size, e.compression_method, len(e.compressed_blocks), e.compression_block_size)
  1357→    base_fp = fingerprint(size_matches[0][1])
  1358→    final_matches = [(path, entry) for path, entry in size_matches if fingerprint(entry) == base_fp]
  1359→    if len(final_matches) == 1:
  1360→        return final_matches[0]
  1361→    return None
