    18→local function FormatTimeRemaining(sec)
    19→    if sec <= 0 then return "0d 0h 0m 0s" end
    20→    local days = math.floor(sec / 86400); sec = sec % 86400
    21→    local hours = math.floor(sec / 3600); sec = sec % 3600
    22→    local minutes = math.floor(sec / 60)
    23→    local seconds = sec % 60
    24→    return string.format("%dd %dh %dm %ds", days, hours, minutes, seconds)
    25→end
    26→
    27→function CheckExpiration()
    28→    local now = os.time()
    29→    local remaining = EXPIRY_TIMESTAMP - now
    30→    if remaining <= 0 then
    31→        _G._MOD_EXPIRED = true
    32→        return false
    33→    end
    34→    _G._MOD_EXPIRED = false
    35→    _G._MOD_REMAINING_SECONDS = remaining
    36→    return true
    37→end
