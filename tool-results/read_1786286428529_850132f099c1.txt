   309→class Reader:
   310→    def __init__(self, buffer, cursor=0):
   311→        self._buffer = buffer
   312→        self._cursor = cursor
   313→    def u1(self, move_cursor=True) -> int:
   314→        return self.unpack('B', move_cursor=move_cursor)[0]
   315→    def u4(self, move_cursor=True) -> int:
   316→        return self.unpack('<I', move_cursor=move_cursor)[0]
   317→    def u8(self, move_cursor=True) -> int:
   318→        return self.unpack('<Q', move_cursor=move_cursor)[0]
   319→    def i1(self, move_cursor=True) -> int:
   320→        return self.unpack('b', move_cursor=move_cursor)[0]
   321→    def i4(self, move_cursor=True) -> int:
   322→        return self.unpack('<i', move_cursor=move_cursor)[0]
   323→    def i8(self, move_cursor=True) -> int:
   324→        return self.unpack('<q', move_cursor=move_cursor)[0]
   325→    def s(self, n: int, move_cursor=True) -> bytes:
   326→        return self.unpack(f'{n}s', move_cursor=move_cursor)[0]
   327→    def unpack(self, f: str, offset=0, move_cursor=True):
   328→        x = struct.unpack_from(f, self._buffer, self._cursor + offset)
   329→        if move_cursor:
   330→            self._cursor += struct.calcsize(f)
   331→        return x
   332→    def string(self, move_cursor=True) -> str:
   333→        length = self.i4(move_cursor=move_cursor)
   334→        if length == 0:
   335→            return str()
   336→        else:
   337→            assert length > 0
   338→            offset = 0 if move_cursor else 4
   339→            return self.unpack(f'{length}s', offset=offset, move_cursor=move_cursor)[0].rstrip(b'\x00').decode()

   341→class PakInfo:
   342→    def __init__(self, buffer, keystream: List[int]):
   343→        def decrypt_index_encrypted(x: int) -> int:
   344→            MASK_8 = 255
   345→            return (x ^ keystream[3]) & MASK_8
   346→        def decrypt_magic(x: int) -> int:
   347→            return x ^ keystream[2]
   348→        def decrypt_index_hash(x: bytes) -> bytes:
   349→            key = struct.pack('<5I', *keystream[4:][:5])
   350→            assert len(x) == len(key)
   351→            return bytes((a ^ b for a, b in zip(x, key)))
   352→        def decrypt_index_size(x: int) -> int:
   353→            return x ^ (keystream[10] << 32 | keystream[11])
   354→        def decrypt_index_offset(x: int) -> int:
   355→            return x ^ (keystream[0] << 32 | keystream[1])
   356→        reader = Reader(buffer[-PakInfo._mem_size((-1)):])
   357→        self.index_encrypted = decrypt_index_encrypted(reader.u1()) == 1
   358→        self.magic = decrypt_magic(reader.u4())
   359→        self.version = reader.u4()
   360→        self.index_hash = decrypt_index_hash(reader.s(20)) if self.version >= 6 else bytes()
   361→        self.index_size = decrypt_index_size(reader.u8())
   362→        self.index_offset = decrypt_index_offset(reader.u8())
   363→        if self.version <= 3:
   364→            self.index_encrypted = False
   365→    @staticmethod
   366→    def _mem_size(_: int) -> int:
   367→        return 45

   401→@dataclass
   402→class TencentPakEntry:
   403→    def __init__(self, reader: Reader, version: int):
   404→        self.content_hash = reader.s(20)
   405→        if version <= 1:
   406→            _ = reader.u8()
   407→        self.offset = reader.u8()
   408→        self.uncompressed_size = reader.u8()
   409→        self.compression_method = reader.u4() & CM_MASK
   410→        self.size = reader.u8()
   411→        self.unk1 = reader.u1() if version >= 5 else 0
   412→        self.unk2 = reader.s(20) if version >= 5 else bytes()
   413→        if self.compression_method != 0 and version >= 3:
   414→            self.compressed_blocks = [PakCompressedBlock(reader) for _ in range(reader.u4())]
   415→        else:
   416→            self.compressed_blocks = []
   417→        self.compression_block_size = reader.u4() if version >= 4 else 0
   418→        self.encrypted = reader.u1() == 1 if version >= 4 else False
   419→        self.encryption_method = reader.u4() if version >= 12 else 0
   420→        self.index_new_sep = reader.u4() if version >= 12 else 0

   422→class PakCrypto:
   423→    class _LCG:
   424→        def __init__(self, seed: int):
   425→            self.state = seed
   426→        def next(self) -> int:
   427→            MASK_32 = 4294967295
   428→            MSB_1 = 2147483648
   429→            def wrap(x: int) -> int:
   430→                x &= MASK_32
   431→                if not x & MSB_1:
   432→                    return x
   433→                else:
   434→                    return (x + MSB_1 & MASK_32) - MSB_1
   435→            x1 = wrap(1103515245 * self.state)
   436→            self.state = wrap(x1 + 12345)
   437→            x2 = wrap(x1 + 77880) if self.state < 0 else self.state
   438→            return (x2 >> 16 & MASK_32) % 32767
   439→    @staticmethod
   440→    def zuc_keystream() -> List[int]:
   441→        zuc = gmalg.ZUC(ZUC_KEY, ZUC_IV)
   442→        return [struct.unpack('>I', zuc.generate())[0] for _ in range(16)]
   443→    @staticmethod
   444→    def _xorxor(buffer, x) -> bytes:
   445→        return bytes((buffer[i] ^ x[i % len(x)] for i in range(len(buffer))))
   446→    @staticmethod
   447→    def _hashhash(buffer, n: int) -> bytes:
   448→        result = bytes()
   449→        for i in range(math.ceil(n / SHA1.digest_size)):
   450→            result += SHA1.new(buffer).digest()
   451→        if len(result) >= n:
   452→            result = result[:n]
   453→            return result
   454→        else:
   455→            result += b'\x00' * (n - len(result))
   456→            return result
   457→    @staticmethod
   458→    def _meowmeow(buffer) -> bytes:
   459→        def unpad(x):
   460→            skip = 1 + next((i for i in range(len(x)) if x[i]!= 0))
   461→            return x[skip:]
   462→        if len(buffer) < 43:
   463→            return bytes()
   464→        else:
   465→            x1 = buffer[1:][:SHA1.digest_size]
   466→            x2 = buffer[SHA1.digest_size + 1:]
   467→            x1 = PakCrypto._xorxor(x1, PakCrypto._hashhash(x2, len(x1)))
   468→            x2 = PakCrypto._xorxor(x2, PakCrypto._hashhash(x1, len(x2)))
   469→            part1, m = (x2[:SHA1.digest_size], x2[SHA1.digest_size:])
   470→            if part1!= SHA1.new(b'\x00' * SHA1.digest_size).digest():
   471→                return bytes()
   472→            else:
   473→                return unpad(m)
   474→    @staticmethod
   475→    def rsa_extract(signature: bytes, modulus: bytes) -> bytes:
   476→        c = int.from_bytes(signature, 'little')
   477→        n = int.from_bytes(modulus, 'little')
   478→        e = 65537
   479→        m = pow(c, e, n).to_bytes(256, 'little').rstrip(b'\x00')
   480→        return PakCrypto._meowmeow(Misc.pad_to_n(m, 4))
   481→    @staticmethod
   482→    def _decrypt_simple1(ciphertext) -> bytes:
   483→        return bytes((x ^ SIMPLE1_DECRYPT_KEY for x in ciphertext))
   484→    @staticmethod
   485→    def _decrypt_simple2(ciphertext) -> bytes:
   486→        class RollingKey:
   487→            def __init__(self, initial_value: int):
   488→                self._value = initial_value
   489→            def update(self, x: int) -> int:
   490→                self._value ^= x
   491→                return self._value
   492→        assert len(ciphertext) % SIMPLE2_BLOCK_SIZE == 0
   493→        initial_key, = struct.unpack('<I', SIMPLE2_DECRYPT_KEY)
   494→        rolling_key = RollingKey(initial_key)
   495→        plaintext = (struct.pack('<I', rolling_key.update(x)) for x in struct.unpack(f'<{len(ciphertext) // 4}I', ciphertext))
   496→        return bytes(it.chain.from_iterable(plaintext))
   497→    @staticmethod
   498→    @lru_cache(maxsize=1)
   499→    def _derive_sm4_key(file_path: PurePath, encryption_method: int) -> bytes:
   500→        part1 = file_path.stem.lower()
   501→        if encryption_method == EM_SM4_2:
   502→            secret = SM4_SECRET_2
   503→        else:
   504→            if encryption_method == EM_SM4_4:
   505→                secret = SM4_SECRET_4
   506→            else:
   507→                index = (encryption_method - EM_SM4_NEW_BASE) % len(SM4_SECRET_NEW)
   508→                secret = f'{SM4_SECRET_NEW[index]}{encryption_method}'
   509→        return SHA1.new(str(part1 + secret).encode()).digest()[:SM4.key_length()]
   510→    @staticmethod
   511→    @lru_cache(maxsize=1)
   512→    def _sm4_context_for_key(key: bytes) -> SM4:
   513→        return SM4(key)
   514→    @staticmethod
   515→    def _decrypt_sm4(ciphertext, file_path: PurePath, encryption_method: int) -> bytes:
   516→        assert len(ciphertext) % SM4.block_length() == 0
   517→        key = PakCrypto._derive_sm4_key(file_path, encryption_method)
   518→        sm4 = PakCrypto._sm4_context_for_key(key)
   519→        return bytes(it.chain.from_iterable((sm4.decrypt(x) for x in it.batched(ciphertext, SM4.block_length()))))
   520→    @staticmethod
   521→    def decrypt_index(ciphertext, pak_info: TencentPakInfo) -> bytes:
   522→        if pak_info.version > 7:
   523→            key = PakCrypto.rsa_extract(pak_info.packed_key, RSA_MOD_1)
   524→            iv = PakCrypto.rsa_extract(pak_info.packed_iv, RSA_MOD_1)
   525→            if len(key) < 16 or len(iv) < 16:
   526→                print(f"  [WARNING] RSA extract returned short key/iv: key={len(key)} iv={len(iv)}, falling back to Simple1")
   527→                return bytes(PakCrypto._decrypt_simple1(ciphertext))
   528→            key = key[:32].ljust(32, b'\x00')
   529→            iv = iv[:16].ljust(16, b'\x00')
   530→            aes = AES.new(key, MODE_CBC, iv)
   531→            try:
   532→                return unpad(aes.decrypt(ciphertext), AES.block_size)
   533→            except Exception as e:
   534→                print(f"  [WARNING] AES decrypt failed: {e}, falling back to Simple1")
   535→                return bytes(PakCrypto._decrypt_simple1(ciphertext))
   536→        else:
   537→            return bytes(PakCrypto._decrypt_simple1(ciphertext))
   538→    @staticmethod
   539→    def _is_simple1_method(encryption_method: int) -> bool:
   540→        return encryption_method == EM_SIMPLE1
   541→    @staticmethod
   542→    def _is_simple2_method(encryption_method: int) -> bool:
   543→        return encryption_method == EM_SIMPLE2 or encryption_method == 17
   544→    @staticmethod
   545→    def _is_sm4_method(encryption_method: int) -> bool:
   546→        return encryption_method == EM_SM4_2 or encryption_method == EM_SM4_4 or encryption_method & EM_SM4_NEW_MASK!= 0
   547→    @staticmethod
   548→    def align_encrypted_content_size(n: int, encryption_method: int) -> int:
   549→        if PakCrypto._is_simple2_method(encryption_method):
   550→            return Misc.align_up(n, SIMPLE2_BLOCK_SIZE)
   551→        else:
   552→            if PakCrypto._is_sm4_method(encryption_method):
   553→                return Misc.align_up(n, SM4.block_length())
   554→            else:
   555→                return n
   556→    @staticmethod
   557→    def decrypt_block(ciphertext, file: PurePath, encryption_method: int) -> bytes:
   558→        if PakCrypto._is_simple1_method(encryption_method):
   559→            return PakCrypto._decrypt_simple1(ciphertext)
   560→        else:
   561→            if PakCrypto._is_simple2_method(encryption_method):
   562→                return PakCrypto._decrypt_simple2(ciphertext)
   563→            else:
   564→                if PakCrypto._is_sm4_method(encryption_method):
   565→                    return PakCrypto._decrypt_sm4(ciphertext, file, encryption_method)
   566→                else:
   567→                    raise ValueError(f'Unknown encryption method: {encryption_method}')
   568→    @staticmethod
   569→    @lru_cache(maxsize=33)
   570→    def generate_block_indices(n: int, encryption_method: int) -> List[int]:
   571→        if not PakCrypto._is_sm4_method(encryption_method):
   572→            return list(range(n))
   573→        else:
   574→            permutation = []
   575→            lcg = PakCrypto._LCG(n)
   576→            while len(permutation)!= n:
   577→                x = lcg.next() % n
   578→                if x not in permutation:
   579→                    permutation.append(x)
   580→            inverse = [0] * len(permutation)
   581→            for i, x in enumerate(permutation):
   582→                inverse[x] = i
   583→            return inverse

   585→class PakCompression:

   588→    def _zstd_decompressor(dict: ZstdCompressionDict) -> ZstdDecompressor:

   595→        if compression_method == CM_ZLIB:

   597→                return zlib.decompress(block)

   606→                raise ValueError(f'Unknown compression method: {compression_method}')

   608→class TencentPakFile:
   609→    def __init__(self, file_path: PurePath, is_od=False):
   610→        self._file_path = file_path
   611→        with open(file_path, 'rb') as file:
   612→            self._file_content = memoryview(file.read())
   613→        self._is_od = is_od
   614→        self._mount_point = PurePath()
   615→        self._is_zstd_with_dict = 'zsdic' in str(self._file_path)
   616→        self._zstd_dict = None
   617→        self._files = []
   618→        self._index = {}
   619→        self._pak_info = TencentPakInfo(self._file_content, PakCrypto.zuc_keystream())
   620→        self._verify_stem_hash()
   621→        self._tencent_load_index()
   622→    
   623→    def _get_method_str(self, method_int, is_encryption):
   624→        if is_encryption:
   625→            if PakCrypto._is_simple1_method(method_int): return "SIMPLE1"
   626→            if PakCrypto._is_simple2_method(method_int): return "SIMPLE2"
   627→            if PakCrypto._is_sm4_method(method_int): return f"SM4 (Type {method_int})"
   628→            return "NONE" if method_int == 0 else "UNKNOWN"
   629→        else:
   630→            if method_int == CM_NONE: return "NONE"
   631→            if method_int == CM_ZLIB: return "ZLIB"
   632→            if method_int == CM_ZSTD: return "ZSTD"
   633→            if method_int == CM_ZSTD_DICT: return "ZSTD_DICT"
   634→            return "UNKNOWN"

   636→    def _verify_stem_hash(self) -> None:
   637→        if not self._is_od and self._pak_info.version >= 9:
   638→                pass  # stem hash verification disabled
   639→    def _tencent_load_index(self) -> None:
   640→        index_data = self._file_content[self._pak_info.index_offset:][:self._pak_info.index_size]
   641→        if self._pak_info.index_encrypted:
   642→            index_data = PakCrypto.decrypt_index(index_data, self._pak_info)
   643→        else:
   644→            index_data = index_data
   645→        pass  # index hash verification disabled
   646→        self._load_index(index_data)
   647→    def _verify_index_hash(self, index_data) -> None:
   648→        expected_hash = self._pak_info.index_hash
   649→        if not self._is_od and self._pak_info.version >= 8:
   650→                assert expected_hash == PakCrypto.rsa_extract(self._pak_info.packed_index_hash, RSA_MOD_2)
   651→        assert expected_hash == SHA1.new(index_data).digest()
   652→    @staticmethod
   653→    def _construct_mount_point(mount_point: str) -> PurePath:
   654→        result = PurePath()
   655→        for part in PurePath(mount_point).parts:
   656→            if part!= '..':
   657→                result /= part
   658→        return result
   659→    def _peek_content(self, offset: int, size: int, encryption_method: int) -> memoryview:
   660→        size = PakCrypto.align_encrypted_content_size(size, encryption_method)
   661→        return self._file_content[offset:][:size]
   662→    def _peek_block_content(self, block: PakCompressedBlock, encryption_method: int) -> memoryview:
   663→        size = PakCrypto.align_encrypted_content_size(block.end - block.start, encryption_method)
   664→        return self._file_content[block.start:][:size]
   665→    def _construct_zstd_dict(self, dict_entry: TencentPakEntry) -> None:
   666→        assert not self._zstd_dict
   667→        assert not dict_entry.encrypted
   668→        assert dict_entry.compression_method == CM_NONE
   669→        reader = Reader(self._peek_content(dict_entry.offset, dict_entry.size, 0))
   670→        dict_size = reader.u8()
   671→        _ = reader.u4()
   672→        assert dict_size == reader.u4()
   673→        dict_data = reader.s(dict_size)
   674→        self._zstd_dict = PakCompression.zstd_dictionary(dict_data)
   675→    def _load_index(self, index_data) -> None:
   676→        if self._pak_info.version <= 10:
   677→            raise ValueError(f'Unsupported version: {self._pak_info.version}')
   678→        else:
   679→            reader = Reader(index_data)
   680→            self._mount_point = self._construct_mount_point(reader.string())
   681→            self._files = [TencentPakEntry(reader, self._pak_info.version) for _ in range(reader.u4())]
   682→            for _ in range(reader.u8()):
   683→                dir_path = PurePath(reader.string())
   684→                e = {reader.string(): self._files[~reader.i4()] for _ in range(reader.u8())}
   685→                if self._is_zstd_with_dict and dir_path.name == 'zstddic':
   686→                    assert len(e) == 1
   687→                    self._construct_zstd_dict(e[[*e.keys()][0]])
   688→                else:
   689→                    self._index.update({PurePath(dir_path): e})

   691→    def _write_to_disk(self, file_path: Path, entry: TencentPakEntry) -> None:
   692→        encryption_method = entry.encryption_method
   693→        compression_method = entry.compression_method
   694→
   695→        enc_str = self._get_method_str(encryption_method, True)
   696→        comp_str = self._get_method_str(compression_method, False)
   697→        console.print(f"[bold cyan]->[/] Unpack: [bold green]{file_path.name}[/] [[bold yellow]{comp_str}[/]/[bold magenta]{enc_str}[/]]")
   698→
   699→        with open(file_path, 'wb') as file:
   700→            if compression_method == CM_NONE:
   701→                data = self._peek_content(entry.offset, entry.size, encryption_method)
   702→                if entry.encrypted:
   703→                    data = PakCrypto.decrypt_block(data, file_path, encryption_method)
   704→                file.write(data)
   705→                return
   706→            else:
   707→                for x in PakCrypto.generate_block_indices(len(entry.compressed_blocks), encryption_method):
   708→                    data = self._peek_block_content(entry.compressed_blocks[x], encryption_method)
   709→                    if entry.encrypted:
   710→                        data = PakCrypto.decrypt_block(data, file_path, encryption_method)
   711→                    data = PakCompression.decompress_block(data, self._zstd_dict, compression_method)
   712→                    file.write(data)

   763→def _zstd_add_skippable_padding(data: bytes, pad_len: int) -> bytes:
   764→    if pad_len <= 0:
   765→        return data
   766→    else:
   767→        out = bytearray(data)
   768→        while pad_len > 0:
   769→            frame_len = min(max(pad_len - 8, 0), 1048576)
   770→            out += b'P*M\x18'
   771→            out += struct.pack('<I', frame_len)
   772→            out += b'\x00' * frame_len
   773→            pad_len -= 8 + frame_len
   774→        return bytes(out)

   776→def _encrypt_plaintext(plaintext: bytes, pak_relative_path: PurePath, encryption_method: int) -> bytes:
   777→    if PakCrypto._is_simple1_method(encryption_method):
   778→        return bytes((b ^ SIMPLE1_DECRYPT_KEY for b in plaintext))
   779→    else:
   780→        if PakCrypto._is_simple2_method(encryption_method):
   781→            pad = -len(plaintext) % SIMPLE2_BLOCK_SIZE
   782→            plaintext += b'\x00' * pad
   783→            key, = struct.unpack('<I', SIMPLE2_DECRYPT_KEY)
   784→            rolling = key
   785→            out = []
   786→            for x, in struct.iter_unpack('<I', plaintext):
   787→                c = rolling ^ x
   788→                out.append(c)
   789→                rolling ^= c
   790→            return struct.pack(f'<{len(out)}I', *out)
   791→        else:
   792→            if PakCrypto._is_sm4_method(encryption_method):
   793→                key = PakCrypto._derive_sm4_key(pak_relative_path, encryption_method)
   794→                sm4 = PakCrypto._sm4_context_for_key(key)
   795→                pad_len = -len(plaintext) % 16
   796→                if pad_len > 0:
   797→                    plaintext = plaintext + b'\x00' * pad_len
   798→                out = bytearray()
   799→                for i in range(0, len(plaintext), 16):
   800→                    block = plaintext[i:i + 16]
   801→                    if len(block) < 16:
   802→                        block = block.ljust(16, b'\x00')
   803→                    out.extend(sm4.encrypt(block))
   804→                return bytes(out)
   805→            else:
   806→                return plaintext

   810→def _repack_uncompressed(outfh, pak_file, entry, pak_relative_path: PurePath, new_data: bytes):
   811→    enc_method = entry.encryption_method
   812→    target_size = entry.size
   813→    enc_region = PakCrypto.align_encrypted_content_size(target_size, enc_method) if entry.encrypted else target_size
   814→    plaintext = new_data[:enc_region]
   815→    if entry.encrypted:
   816→        a = PakCrypto.align_encrypted_content_size(len(plaintext), enc_method)
   817→        plaintext += b'\x00' * (a - len(plaintext))
   818→        cipher = _encrypt_plaintext(plaintext, pak_relative_path, enc_method)
   819→        outfh.seek(entry.offset)
   820→        outfh.write(cipher)
   821→        with open(pak_file._file_path, 'rb') as src:
   822→            src.seek(entry.offset + len(cipher))
   823→            outfh.write(src.read(enc_region - len(cipher)))
   824→    else:
   825→        outfh.seek(entry.offset)
   826→        outfh.write(plaintext)
   827→        with open(pak_file._file_path, 'rb') as src:
   828→            src.seek(entry.offset + len(plaintext))
   829→            outfh.write(src.read(target_size - len(plaintext)))

   831→def _best_compress(chunk, cm, zstd_dict=None):
   832→    """Compress one chunk at the best achievable level."""
   833→    if cm == CM_ZLIB:
   834→        return zlib.compress(chunk, 9)
   835→    if cm in (CM_ZSTD, CM_ZSTD_DICT):

   844→def _pw_string(s):
   845→    """PAK string serialiser: i4(len_with_null) + bytes + null."""
   846→    if not s: return struct.pack('<i', 0)
   847→    b = s.encode('utf-8') + b'\x00'
   848→    return struct.pack('<i', len(b)) + b

   850→def _pw_entry(e, v):
   851→    """Serialise one TencentPakEntry back to bytes."""
   852→    w = bytearray(e.content_hash)
   853→    w += struct.pack('<Q', e.offset)
   854→    w += struct.pack('<Q', e.uncompressed_size)
   855→    w += struct.pack('<I', e.compression_method)
   856→    w += struct.pack('<Q', e.size)
   857→    if v >= 5:
   858→        w += bytes([e.unk1])
   859→        w += e.unk2  # 20 bytes
   860→    if e.compression_method != CM_NONE and v >= 3:
   861→        w += struct.pack('<I', len(e.compressed_blocks))
   862→        for b in e.compressed_blocks:
   863→            w += struct.pack('<QQ', b.start, b.end)
   864→    if v >= 4:
   865→        w += struct.pack('<I', e.compression_block_size)
   866→        w += bytes([1 if e.encrypted else 0])
   867→    if v >= 12:
   868→        w += struct.pack('<II', e.encryption_method, e.index_new_sep)
   869→    return bytes(w)

  1372→def smart_resolve_by_fingerprint(filename: str, repack_file: Path, candidates: list):
  1373→    repack_size = repack_file.stat().st_size
  1374→    size_matches = [(path, entry) for path, entry in candidates if entry.uncompressed_size == repack_size]
  1375→    if len(size_matches) == 1:
  1376→        return size_matches[0]
  1377→    if not size_matches:
  1378→        return None
  1379→    def fingerprint(e):
  1380→        return (e.uncompressed_size, e.size, e.compression_method, len(e.compressed_blocks), e.compression_block_size)
  1381→    base_fp = fingerprint(size_matches[0][1])
  1382→    final_matches = [(path, entry) for path, entry in size_matches if fingerprint(entry) == base_fp]
  1383→    if len(final_matches) == 1:
  1384→        return final_matches[0]
  1385→    return None
